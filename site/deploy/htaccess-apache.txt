# Apache — facturexreader.com
#
# À placer dans le répertoire racine du site (à côté de index.html).
# Le module mod_rewrite et l'AllowOverride All doivent être actifs.

Options -Indexes
DirectoryIndex index.html

ErrorDocument 404 /404.html

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml \
    application/javascript application/json application/xml image/svg+xml
</IfModule>

<IfModule mod_expires.c>
  ExpiresActive On

  # Ressources dont le contenu change rarement : cache long.
  <FilesMatch "\.(webp|jpg|png|svg|woff2|ttf)$">
    ExpiresDefault "access plus 30 days"
    Header set Cache-Control "public, max-age=2592000"
  </FilesMatch>

  # Le HTML se revalide à chaque visite.
  <FilesMatch "\.html$">
    ExpiresDefault "access plus 0 seconds"
    Header set Cache-Control "public, must-revalidate"
  </FilesMatch>
</IfModule>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
  # Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
</IfModule>

# URL propres : /mentions-legales -> mentions-legales.html
